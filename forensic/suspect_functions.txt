def strip_screenshot_code_from_build_terrain():
    target = SCRIPTS_DIR / "build_terrain.gd"
    if not target.exists():
        print("[STRIP-ERROR] build_terrain.gd not found", flush=True)
        return False

    content = target.read_text()
    original = content

    # Remove lines containing "save_png"
    lines = content.splitlines()
    new_lines = []
    removed = 0
    for line in lines:
        if "save_png" in line:
            print(f"[STRIP] Removing line: {line.strip()}", flush=True)
            removed += 1
            continue
        new_lines.append(line)
    if removed == 0:
        print("[STRIP] No screenshot code found in build_terrain.gd", flush=True)
        # Still run gdformat to ensure indentation is clean
        subprocess.run(["gdformat", str(target)], check=False)
        print("[STRIP] gdformat applied (no changes needed).", flush=True)
        return True

    new_content = "\n".join(new_lines)

    # Aggressive removal if any remained
    if "save_png" in new_content:
        print("[STRIP-ERROR] save_png still present after stripping!", flush=True)
        new_content = re.sub(r'^.*save_png.*$', '', new_content, flags=re.MULTILINE)
        new_content = re.sub(r'\n\s*\n', '\n', new_content)

    if "save_png" in new_content:
        print("[STRIP-FATAL] Could not remove save_png line. Manual intervention required.", flush=True)
        return False

    # Add headless guard if not present
    if "if Engine.is_in_headless_mode():" not in new_content:
        new_content = re.sub(
            r'(func _process\s*\([^)]*\):\s*)',
            r'\1\n\tif Engine.is_in_headless_mode(): return',
            new_content
        )
        print("[STRIP] Added headless guard.", flush=True)

    # Write the stripped content
    target.write_text(new_content)
    _r082_verify(target, new_content, "build_terrain.gd stripped")

    # Run gdformat to fix indentation after line removal (FIX-167-001)
    subprocess.run(["gdformat", str(target)], check=False)
    print("[STRIP] gdformat applied to fix indentation.", flush=True)

    # Verify the file still passes gdparse
    result = subprocess.run(["gdparse", str(target)], capture_output=True, text=True)
    if result.returncode != 0:
        print("[STRIP-ERROR] File fails gdparse after stripping and formatting.", flush=True)
        print(result.stdout + result.stderr, flush=True)
        return False

    print("[STRIP] Successfully removed save_png, added headless guard, and fixed indentation.", flush=True)
    return True

def gate_build_terrain_preflight() -> bool:
    bt = SCRIPTS_DIR / "build_terrain.gd"
    if not bt.exists():
        print("[GATE-FAIL] build_terrain.gd not found", flush=True)
        return False

    content = bt.read_text()
    line_count = len(content.splitlines())
    print(f"[GATE] build_terrain.gd: {line_count} lines", flush=True)

    result = subprocess.run(["gdparse", str(bt)], capture_output=True, text=True)
    if result.returncode != 0:
        print("[GATE-FAIL] build_terrain.gd fails gdparse. Attempting restore from backup...", flush=True)
        if restore_build_terrain_from_backup():
            print("[GATE-RESTORE] Restore successful. Re-checking...", flush=True)
            result2 = subprocess.run(["gdparse", str(bt)], capture_output=True, text=True)
            if result2.returncode == 0:
                print("[GATE-OK] build_terrain.gd passes gdparse after restore.", flush=True)
                # Now strip screenshot code if present
                content = bt.read_text()
                if "save_png" in content:
                    print("[GATE-WARN] save_png still present after restore – stripping.", flush=True)
                    return strip_screenshot_code_from_build_terrain()
                return True
            else:
                print("[GATE-FAIL] build_terrain.gd still fails gdparse after restore.", flush=True)
                print(result2.stdout + result2.stderr, flush=True)
                return False
        else:
            print("[GATE-FAIL] Restore failed.", flush=True)
            return False

    print("[GATE-OK] build_terrain.gd passes gdparse", flush=True)

    if "save_png" in content:
        print("[GATE-WARN] build_terrain.gd contains 'save_png' – will strip it.", flush=True)
        if not strip_screenshot_code_from_build_terrain():
            return False
        # Re-read after strip and verify
        content = bt.read_text()
        if "save_png" in content:
            print("[GATE-FAIL] save_png still present after strip.", flush=True)
            return False
    print("[GATE-OK] build_terrain.gd: no screenshot code present — correct", flush=True)

    return True
