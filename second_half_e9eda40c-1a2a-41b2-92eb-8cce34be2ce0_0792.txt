        # Empty result is still a success (e.g., INSERT)

func register_operation(operation_id: String, recovery_func: Callable, max_retries: int = MAX_RETRIES):
    """Register an operation that can be recovered"""
    _recovery_operations[operation_id] = {
        "func": recovery_func,
        "max_retries": max_retries,
        "attempts": 0,
        "last_attempt": 0,
        "delay_ms": BASE_DELAY_MS
    }
    print("[VERBATIM] RecoveryManager: registered operation '%s'" % operation_id)

func attempt_recovery(operation_id: String) -> bool:
    """Attempt to recover a specific operation"""
    if _is_recovering:
        print("[VERBATIM] RecoveryManager: already recovering, skipping")
        return false
    
    var op = _recovery_operations.get(operation_id)
    if not op:
        print("[VERBATIM] RecoveryManager: operation '%s' not registered" % operation_id)
        return false
    
    # Check throttle
    if _throttle and not _throttle.can_attempt("recover_" + operation_id):
        print("[VERBATIM] RecoveryManager: operation '%s' throttled" % operation_id)
        return false
    
    _is_recovering = true
    
    # Check if database is healthy
    if _db_guard and not _db_guard.is_healthy():
        print("[VERBATIM] RecoveryManager: database unhealthy, waiting")
        _is_recovering = false
        return false
    
    # Check if we need to wait
    var now = Time.get_ticks_msec()
    if now - op["last_attempt"] < op["delay_ms"]:
        print("[VERBATIM] RecoveryManager: operation '%s' cooling down" % operation_id)
        _is_recovering = false
        return false
    
    # Attempt recovery
    op["last_attempt"] = now
    op["attempts"] += 1
    
    var success = false
    try:
        success = op["func"].call()
    except:
        print("[VERBATIM] RecoveryManager: recovery function threw error")
        success = false
    
    # Update throttle
    if _throttle:
        _throttle.record_attempt("recover_" + operation_id, success)
    
    if success:
        # Reset on success
        op["attempts"] = 0
        op["delay_ms"] = BASE_DELAY_MS
        print("[VERBATIM] RecoveryManager: operation '%s' recovered successfully" % operation_id)
    else:
        # Exponential backoff on failure
        op["delay_ms"] = min(op["delay_ms"] * 2, MAX_DELAY_MS)
        print("[VERBATIM] RecoveryManager: operation '%s' recovery failed, backoff %d ms" % [operation_id, op["delay_ms"]])
        
        if op["attempts"] >= op["max_retries"]:
            print("[VERBATIM] RecoveryManager: operation '%s' max retries exceeded" % operation_id)
    
    _is_recovering = false
    return success

func get_status(operation_id: String) -> Dictionary:
    """Get recovery status for an operation"""
    var op = _recovery_operations.get(operation_id)
    if not op:
        return {}
    
    return {
        "operation": operation_id,
        "attempts": op["attempts"],
        "max_retries": op["max_retries"],
        "next_retry_in": max(0, op["delay_ms"] - (Time.get_ticks_msec() - op["last_attempt"]))
    }

func schedule_maintenance():
    """Schedule database maintenance tasks"""
    if not _db_guard:
        return
    
    # Check if we should compact
    var health = _db_guard.get_health_report()
    if health.get("failure_count", 0) > 10:
        print("[VERBATIM] RecoveryManager: high failure count, scheduling compaction")
        _db_guard.vacuum_db()

func can_attempt(operation: String) -> bool:
    var db = get_node_or_null("/root/SqliteDb")
    if not db:
        return true  # Allow if DB not available
    
    var sql = "SELECT attempt_count, backoff_seconds, last_attempt, locked FROM throttle_log WHERE operation = ?"
    var result = db._query(sql, [operation])
    
    if result.is_empty():
        return true
    
    var row = result[0]
    var attempt_count = int(row[0])
    var backoff_seconds = int(row[1])
    var last_attempt = str(row[2])
    var locked = int(row[3])
    
    if locked == 1:
        print("[VERBATIM] ResourceThrottle: operation '%s' is locked" % operation)
        return false
    
    if last_attempt != "" and last_attempt != "null":
        var last_time = Time.get_datetime_dict_from_string(last_attempt)
        var now = Time.get_datetime_dict_from_system()
        var elapsed = _datetime_diff_seconds(last_time, now)
        if elapsed < backoff_seconds:
            return false
    
    return true

func record_attempt(operation: String, success: bool, error_msg: String = ""):
    var db = get_node_or_null("/root/SqliteDb")
    if not db:
        return
    
    # Get current state
    var sql = "SELECT attempt_count, backoff_seconds, failure_count, locked FROM throttle_log WHERE operation = ?"
    var result = db._query(sql, [operation])
    
    var attempt_count = 1
    var backoff_seconds = 30
    var failure_count = 0
    var locked = 0
    
    if not result.is_empty():
        var row = result[0]
        attempt_count = int(row[0]) + 1
        backoff_seconds = int(row[1])
        failure_count = int(row[2])
        locked = int(row[3])
    
    if success:
        failure_count = 0
        backoff_seconds = max(10, backoff_seconds / 2)
        locked = 0
    else:
        failure_count += 1
        if failure_count > 3:
            backoff_seconds = min(3600, backoff_seconds * 2)
        if failure_count > 5:
            locked = 1
            print("[VERBATIM] ResourceThrottle: operation '%s' LOCKED due to repeated failures" % operation)
    
    var now = Time.get_datetime_string_from_system()
    
    sql = """
        INSERT OR REPLACE INTO throttle_log 
        (operation, attempt_count, last_attempt, backoff_seconds, success_count, failure_count, locked, last_error)
        VALUES (?, ?, ?, ?, ?, ?, ?, ?)
    """
    db._query(sql, [
        operation,
        attempt_count,
        now,
        backoff_seconds,
        1 if success else 0,
        failure_count,
        locked,
        error_msg
    ])

func _datetime_diff_seconds(a: Dictionary, b: Dictionary) -> int:
    # Calculate seconds difference between two datetime dicts
    var a_sec = a.get("year", 0) * 31536000 + a.get("month", 0) * 2592000 + a.get("day", 0) * 86400 + a.get("hour", 0) * 3600 + a.get("minute", 0) * 60 + a.get("second", 0)
    var b_sec = b.get("year", 0) * 31536000 + b.get("month", 0) * 2592000 + b.get("day", 0) * 86400 + b.get("hour", 0) * 3600 + b.get("minute", 0) * 60 + b.get("second", 0)
    return abs(b_sec - a_sec)

func _monitor_loop():
    print("[VERBATIM] PerformanceMonitor: _monitor_loop started")
    var interval: int = _monitor_interval_ms
    
    while not _stop:
        # Check system load
        var load = _read_load_avg()
        if load > 2.0:
            interval = min(interval * 2, _max_interval_ms)
            if interval >= _max_interval_ms:
                print("[VERBATIM] PerformanceMonitor: high load, max throttling")
        elif interval > _monitor_interval_ms:
            interval = max(_monitor_interval_ms, interval / 2)
        
        var cpu = _read_cpu_usage()
        var mem = _read_memory()
        var cs = _read_context_switches()
        
        if _db_ok and _db_node:
            var success = _insert_contention(_pid, cpu, mem, cs)
            if success:
                _consecutive_failures = 0
                if _failure_count > 0:
                    _failure_count -= 1
            else:
                _consecutive_failures += 1
                _failure_count += 1
                if _consecutive_failures > 5:
                    print("[VERBATIM] PerformanceMonitor: DB failures, throttling")
                    interval = min(interval * 2, _max_interval_ms)
                    _consecutive_failures = 0
        
        # Sleep with smaller chunks to respond to _stop faster
        var sleep_chunk = min(interval, 1000)
        var slept = 0
        while not _stop and slept < interval:
            OS.delay_msec(sleep_chunk)
            slept += sleep_chunk

func _read_load_avg() -> float:
    var file = FileAccess.open("/proc/loadavg", FileAccess.READ)
    if not file:
        return 0.0
    var line = file.get_line()
    file.close()
    var parts = line.split(" ")
    if parts.size() >= 1:
        return float(parts[0])
    return 0.0

func _insert_contention(pid: int, cpu: float, mem: int, cs: Dictionary) -> bool:
    if not _db_node:
        return false
    if not _db_node.has_method("_query"):
        return false
    
    # Check throttle before heavy operation
    if _throttle and not _throttle.can_attempt("db_insert_contention"):
        return false
    
    var sql = """
        INSERT INTO resource_contention_log
        (session_id, godot_pid, ts, vm_rss_kb, vol_ctxt_sw, nonvol_ctxt_sw, cpu_pct, io_wait_pct, sched_policy)
        VALUES (?, ?, datetime('now'), ?, ?, ?, ?, ?, ?)
    """
    var args = [24, pid, mem, cs.get("vol", 0), cs.get("nonvol", 0), cpu, 0.0, "SCHED_OTHER"]
    
    var result = _db_node._query(sql, args)
    var success = result.size() > 0
    
    if _throttle:
        _throttle.record_attempt("db_insert_contention", success)

func _read_cpu_usage() -> float:
    var file = FileAccess.open("/proc/self/stat", FileAccess.READ)
    if not file:
        return 0.0
    var line = file.get_line()
    file.close()
    var parts = line.split(" ")
    if parts.size() < 15:
        return 0.0
    var utime = int(parts[13])
    var stime = int(parts[14])
    return min(float(utime + stime) * 0.01, 100.0)

func _read_memory() -> int:
    var file = FileAccess.open("/proc/self/status", FileAccess.READ)
    if not file:
        return 0
    while not file.eof_reached():
        var line = file.get_line()
        if line.begins_with("VmRSS:"):
            var parts = line.split(" ")
            if parts.size() >= 2:
                return int(parts[parts.size() - 2])
    return 0

func _read_context_switches() -> Dictionary:
    var file = FileAccess.open("/proc/self/status", FileAccess.READ)
    if not file:
        return {"vol": 0, "nonvol": 0}
    var vol = 0
    var nonvol = 0
    while not file.eof_reached():
        var line = file.get_line()
        if line.begins_with("voluntary_ctxt_switches:"):
            var parts = line.split(" ")
            if parts.size() >= 2:
                vol = int(parts[parts.size() - 1])
        elif line.begins_with("nonvoluntary_ctxt_switches:"):
            var parts = line.split(" ")
            if parts.size() >= 2:
                nonvol = int(parts[parts.size() - 1])
    return {"vol": vol, "nonvol": nonvol}

Add these functions to your build_terrain.gd:
gdscript

# Add at the top with other variables
var _db_check_timer: float = 0.0
var _db_check_interval: float = 5.0  # Check every 5 seconds
var _last_db_health: bool = true

# Add this function to _ready() after other initialization
func _init_database_guard():
    var guard = get_node_or_null("/root/DatabaseGuard")
    if guard:
        guard.db_healthy.connect(_on_db_healthy)
        guard.db_unhealthy.connect(_on_db_unhealthy)
        guard.db_recovered.connect(_on_db_recovered)
        _log("DatabaseGuard connected")
    else:
        _log("WARNING: DatabaseGuard not found")

# Call this in _ready() after creating the HUD
func _ready():
    # ... existing code ...
    _init_database_guard()
    # ... rest of existing code ...

# Add this function to _process(delta)
func _process(delta):
    # ... existing code ...
    
    # Periodic database health check
    _db_check_timer += delta
    if _db_check_timer >= _db_check_interval:
        _db_check_timer = 0.0
        _check_database_health()
    
    # ... rest of existing code ...

func _check_database_health():
    var guard = get_node_or_null("/root/DatabaseGuard")
    if not guard:
        return
    
    var health = guard.get_health_report()
    var is_healthy = health.get("is_healthy", false)
    
    if is_healthy != _last_db_health:
        _last_db_health = is_healthy
        if is_healthy:
            _log("Database is healthy")
        else:
            _log("Database unhealthy: %s" % health.get("last_error", "unknown"))

func _on_db_healthy():
    _log("Database health: OK")

func _on_db_unhealthy(error: String):
    _log("Database health: UNHEALTHY - %s" % error)

func _on_db_recovered():
    _log("Database health: RECOVERED")

# Add this to _exit_tree() if you have one
func _exit_tree():
    var guard = get_node_or_null("/root/DatabaseGuard")
    if guard:
        _log("DatabaseGuard: closing")
