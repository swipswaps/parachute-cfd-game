     4	# ==========================================================================
     5	# forensic_hud.gd  (autoload as ForensicHUD)
     6	#
     7	# In-game overlay for the forensic HUB. Both this HUD and the browser view
     8	# are HTTP clients of the same forensic hub server; single source of truth.
     9	#
    10	# ARCHITECTURE (do not "simplify" -- each choice fixes a prior real bug):
    11	#   * extends Node, NOT CanvasLayer: get_global_mouse_position() is defined
    12	#     on CanvasItem, not CanvasLayer (prior gdparse failure).
    13	#   * extends Node, NOT Control: the visible panel lives inside a
    14	#     CanvasLayer child, and CanvasLayer ignores its parent's canvas
    15	#     transform -- dragging the root's global_position moves NOTHING on
    16	#     screen. Drag must move _panel.position directly.
    17	#   * Drag uses a rect hit-test on _panel with event.position (viewport
    18	#     coords == CanvasLayer coords, since _layer has no transform).
    19	#     Without the hit-test, every left-click in the game would be
    20	#     swallowed by set_input_as_handled(), breaking the PiP drag too.
    21	#   * F3 toggles _layer.visible (toggling the root's visibility does not
    22	#     hide CanvasLayer children).
    23	#   * No per-mouse-event print() calls: those flooded the log and
    24	#     triggered autostall.py [STALL SOURCE] warnings previously.
    25	#
    26	# Controls: F3 toggles the panel. Left-click + drag anywhere on the panel
    27	# to move it. Position persists to user://forensic_hud.cfg. Each citation
    28	# card has a [+]/[-] toggle; URL rows open via OS.shell_open.
    29	# ==========================================================================

    67	func _resolve_hub_url() -> String:
    68		var key := "application/forensic_hub/url"
    69		if ProjectSettings.has_setting(key):
    70			var v = ProjectSettings.get_setting(key)
    71			if typeof(v) == TYPE_STRING and String(v) != "":
    72				return String(v)
    73		var env := OS.get_environment("FORENSIC_HUB_URL")
    74		if env != "":
    75			return env
    76		return DEFAULT_HUB_URL

   260	) -> void:
   261		_time_label.text = "t: " + Time.get_time_string_from_system()

   271			_status_label.text = "hub OK, DB unavailable"
   272			_gam_label.text = "(no pipeline DB found)"

   274		_status_label.text = "hub OK  (%s)" % _hub_url
   275		var lv = d.get("level", "-")
   276		var xp = d.get("xp", "-")
   277		var st = d.get("streak", "-")
   278		var sr = d.get("success_rate", 0.0)

   280		var known = d.get("pokedex_known", "-")
   281		var unresolved = d.get("unresolved_parse_errors", "-")
   282		var ach = d.get("achievements", [])
   283		var ach_lines := "achievements:"

   289			+ "pokedex: %s / %s trained\n"

   293	func _on_leader_completed(
   294		result: int, response_code: int, _headers: PackedStringArray, body: PackedByteArray
   295	) -> void:
   296		if result != HTTPRequest.RESULT_SUCCESS or response_code != 200:
   297			return
   298		var parsed = JSON.parse_string(body.get_string_from_utf8())
   299		if typeof(parsed) != TYPE_ARRAY:
   300			return
   301		for c in _leader_vbox.get_children():
   302			c.queue_free()
   303		var rows: Array = parsed
   304		if rows.is_empty():
   305			_leader_vbox.add_child(_mklabel("  (no strategies yet)"))
   306			return
   307		var shown := 0
   308		for row in rows:
   309			if shown >= 6:
   310				break
   311			if typeof(row) != TYPE_DICTIONARY:
   312				continue
   313			var r: Dictionary = row
   314			var txt := "  %s [%s] wins=%s/%s wr=%s ucb1=%s" % [
   315				str(r.get("name", "?")),
   316				str(r.get("error_class", "?")),
   317				str(r.get("wins", 0)),
   318				str(r.get("trials", 0)),
   319				str(r.get("win_rate", 0)),
   320				str(r.get("ucb1", 0)),
   321			]
   322			_leader_vbox.add_child(_mklabel(txt))
   323			shown += 1

   325	func _on_cite_completed(
   326		result: int, response_code: int, _headers: PackedStringArray, body: PackedByteArray
   327	) -> void:
   328		if result != HTTPRequest.RESULT_SUCCESS or response_code != 200:
   329			return
   330		var parsed = JSON.parse_string(body.get_string_from_utf8())
   331		if typeof(parsed) != TYPE_DICTIONARY:
   332			return
   333		var d: Dictionary = parsed
   334		var all_cites: Array = []
   335		if d.has("principles") and typeof(d["principles"]) == TYPE_ARRAY:
   336			all_cites += d["principles"]
   337		if d.has("godot_docs") and typeof(d["godot_docs"]) == TYPE_ARRAY:
   338			all_cites += d["godot_docs"]
   339		if all_cites.is_empty():
   340			return
   341		_citations_loaded = true
   342		for c in _cite_vbox.get_children():
   343			c.queue_free()
   344		for cite in all_cites:
   345			if typeof(cite) != TYPE_DICTIONARY:
   346				continue
   347			_add_citation_card(cite as Dictionary)

   363	func _add_citation_card(cite: Dictionary) -> void:
   364		var cid := str(cite.get("id", "?"))
   365		var label := str(cite.get("label", "(unlabeled)"))
   366		var url := str(cite.get("url", ""))
   367		var verbatim := str(cite.get("verbatim", ""))
   368		var applies := str(cite.get("applies_to", ""))
   369		var header_row := HBoxContainer.new()
   370		var toggle := _mkbutton("+")
   371		toggle.custom_minimum_size = Vector2(22, 22)
   372		header_row.add_child(toggle)
   373		header_row.add_child(_mklabel(" [%s] %s" % [cid, label], Color(0.85, 0.95, 0.85, 1.0)))
   374		_cite_vbox.add_child(header_row)
   375	
   376		var body_box := VBoxContainer.new()
   377		body_box.visible = _card_expanded.get(cid, false)
   378		var quote_label := _mklabel("  \"" + verbatim + "\"", Color(0.85, 0.85, 0.85, 1.0))
   379		quote_label.autowrap_mode = TextServer.AUTOWRAP_WORD
   380		body_box.add_child(quote_label)
   381		if applies != "":
   382			var applies_label := _mklabel("  applies: " + applies, Color(0.7, 0.9, 0.7, 1.0))
   383			applies_label.autowrap_mode = TextServer.AUTOWRAP_WORD
   384			body_box.add_child(applies_label)
   385		if url != "":
   386			var url_btn := _mkbutton("  open: " + url)
   387			url_btn.pressed.connect(func(): OS.shell_open(url))
   388			body_box.add_child(url_btn)
   389		_cite_vbox.add_child(body_box)
   390	
   391		toggle.pressed.connect(func():
   392			var cur = _card_expanded.get(cid, false)
   393			_card_expanded[cid] = not cur
   394			body_box.visible = not cur
   395			toggle.text = "-" if not cur else "+"
   396		)

   399	func _gui_input(event: InputEvent) -> void:
   400		
   401		print("[HUD_DIAG] _gui_input called")
   402		print("[HUD] _gui_input called: ", event)
   403		# Only process if panel is visible
   404		if _layer == null or not _layer.visible or _panel == null:
   405			return
   406		if event is InputEventMouseButton:
   407			var mb := event as InputEventMouseButton
   408	
   409			if mb.button_index != MOUSE_BUTTON_LEFT:
   410				return
   411			if mb.pressed and _panel_rect().has_point(mb.position):
   412				_dragging = true
   413				_drag_offset = mb.position - _panel.position
   414				get_viewport().set_input_as_handled()
   415			elif not mb.pressed and _dragging:
   416				_dragging = false
   417				get_viewport().set_input_as_handled()
   418		if event is InputEventMouseMotion and _dragging:
   419			_panel.position = event.position - _drag_offset
   420			get_viewport().set_input_as_handled()
   421	
   422	func _confirm_drag_attached() -> void:
   423		if _panel and _panel.has_method("_gui_input"):
   424			print("[VERBATIM] HUD drag script _ready — _gui_input confirmed")
   425		else:
   426			print("[VERBATIM] HUD drag script MISSING — ",
   427				"_panel=", _panel, " script=",
   428				_panel.get_script() if _panel else '(null)')
