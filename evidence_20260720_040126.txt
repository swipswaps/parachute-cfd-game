   53: func _input(event: InputEvent) -> void:

   56: 			panel.visible = not panel.visible
   57: 			get_viewport().set_input_as_handled()
   58: 			return
   59: 
   60: 	if event is InputEventMouseButton:
   61: 		var mb := event as InputEventMouseButton
   62: 		if mb.button_index == MOUSE_BUTTON_LEFT:
   63: 			var pos: Vector2 = mb.position

   66: 				dragging = true

   68: 				get_viewport().set_input_as_handled()
   69: 			elif not mb.pressed:
   70: 				dragging = false
   71: 
   72: 	if event is InputEventMouseMotion and dragging:
   73: 		var mm := event as InputEventMouseMotion
   74: 		panel.position = mm.position - drag_offset
