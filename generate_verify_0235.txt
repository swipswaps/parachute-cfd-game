def fix_sqlitedb_path(conn: sqlite3.Connection) -> bool:
    """
    WHAT: replace 'user://parachute.db' with the absolute path of
          parachute_mutations.db in SqliteDb.gd line 4.
    WHY:  confirmed from STEP 2 verbatim: const DB_PATH := "user://parachute.db"
          This is the root cause of the two-DB problem. One file fix.
    FAILURE MODE: staleness check fails if line 4 has already been changed.
    VERIFIES WITH: grep DB_PATH SqliteDb.gd shows parachute_mutations.db.
    Source (Tier 1): Date (2003) "Introduction to Database Systems" Ch.1 p.7.
    Source (Tier 1): Stevens & Rago (2013) "APUE" Ch.3 p.55.
    Source (Tier 4): https://github.com/2shady4u/godot-sqlite/blob/master/src/gd_sqlite.cpp
    """
    if not SQLITEDB_GD.exists():
        print(f"  [ERROR] {SQLITEDB_GD} not found")
        return False

    content = SQLITEDB_GD.read_text(encoding="utf-8")
    old_line = 'const DB_PATH := "user://parachute.db"'
    new_line = f'const DB_PATH := "{DB_PATH}"'

    if new_line in content:
        print(f"  [SKIP] DB_PATH already points to parachute_mutations.db")
        return True

    if old_line not in content:
        # Check what's actually there
        for i, ln in enumerate(content.splitlines(), 1):
            if "DB_PATH" in ln:
                print(f"  [WARN] DB_PATH at line {i}: {ln!r}")
        print(f"  [ERROR] Expected line not found: {old_line!r}")
        return False

    new_content = content.replace(old_line, new_line, 1)

    ok, sha_before, sha_after = apply_with_structural_gate(
        conn, SESSION_ID, SQLITEDB_GD,
        lambda: SQLITEDB_GD.write_text(new_content, encoding="utf-8")
    )
    if not ok:
        return False

    # Verify
    written = SQLITEDB_GD.read_text(encoding="utf-8")
    if str(DB_PATH) not in written:
        print(f"  [ERROR] Read-back failed — DB_PATH not in written file")
        return False

    conn.execute("""
        INSERT OR IGNORE INTO gd_patch_log
            (session_id, file_path, patch_id_str, lines_changed,
             sha256_before, sha256_after, verified)
        VALUES (?,?,?,?,?,?,1)
    """, (SESSION_ID, str(SQLITEDB_GD), "F24-sqlitedb-absolute-path",
          1, sha_before, sha_after))
    conn.commit()
    print(f"  [OK] SqliteDb.gd DB_PATH -> {DB_PATH}")
    print(f"       sha_before={sha_before[:16]}  sha_after={sha_after[:16]}")
    log_orchestration(conn, "FIX_APPLIED", SQLITEDB_GD, "PASS")
    return True

def _read_proc_status(pid: int) -> dict:
    """
    WHAT: read /proc/PID/status for VmRSS and context switches.
    WHY:  resource_contention_log requires rss_kb, vol_ctxt_sw, nonvol_ctxt_sw.
    FAILURE MODE: process exited -> returns {}.
    Source (Tier 2): Linux procfs: https://www.kernel.org/doc/html/latest/filesystems/proc.html
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.5 p.208.
    """
    try:
        text = Path(f"/proc/{pid}/status").read_text()
        result = {}
        for ln in text.splitlines():
            if ln.startswith("VmRSS:"):
                result["vm_rss_kb"] = int(ln.split()[1])
            elif ln.startswith("voluntary_ctxt_switches:"):
                result["vol_ctxt_sw"] = int(ln.split()[1])
            elif ln.startswith("nonvoluntary_ctxt_switches:"):
                result["nonvol_ctxt_sw"] = int(ln.split()[1])
        return result
    except (FileNotFoundError, PermissionError):
        return {}


def _read_proc_stat() -> list:
    """
    WHAT: read /proc/stat cpu line for CPU/IO-wait calculation.
    WHY:  cpu_pct and io_wait_pct require two samples (delta).
    Source (Tier 2): https://www.kernel.org/doc/html/latest/filesystems/proc.html
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.6 p.258.
    """
    try:
        for ln in Path("/proc/stat").read_text().splitlines():
            if ln.startswith("cpu "):
                return [int(x) for x in ln.split()[1:]]
    except Exception:
        pass
    return [0] * 10


def compute_cpu_pct(prev: list, curr: list) -> tuple[float, float]:
    """
    WHAT: compute CPU% and IO-wait% from two /proc/stat samples.
    WHY:  USE method requires utilisation (CPU%) and saturation (IO wait%).
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.2 p.58
      — USE method: Utilisation, Saturation, Errors.
    Source (Tier 2): https://www.kernel.org/doc/html/latest/filesystems/proc.html
    """
    if len(prev) < 8 or len(curr) < 8:
        return 0.0, 0.0
    # Fields: user nice system idle iowait irq softirq steal
    total_prev = sum(prev[:8])
    total_curr = sum(curr[:8])
    delta = total_curr - total_prev
    if delta == 0:
        return 0.0, 0.0
    idle_prev  = prev[3];  idle_curr  = curr[3]
    iowait_prev = prev[4]; iowait_curr = curr[4]
    cpu_pct    = 100.0 * (1.0 - (idle_curr - idle_prev) / delta)
    io_pct     = 100.0 * (iowait_curr - iowait_prev) / delta
    return round(cpu_pct, 1), round(io_pct, 1)

        WHY:  the main thread is blocked on proc.stdout readline.

        # Own connection — WAL mode allows concurrent reads

            curr = _read_proc_stat()
            cpu_pct, io_pct = compute_cpu_pct(prev_stat, curr)

            try:

                      m.get("nonvol_ctxt_sw"), cpu_pct, io_pct))
                mconn.commit()
            except Exception:
                pass
            if m.get("vm_rss_kb"):

def fix_sqlitedb_path(conn: sqlite3.Connection) -> bool:
    """
    WHAT: replace 'user://parachute.db' with the absolute path of
          parachute_mutations.db in SqliteDb.gd line 4.
    WHY:  confirmed from STEP 2 verbatim: const DB_PATH := "user://parachute.db"
          This is the root cause of the two-DB problem. One file fix.
    FAILURE MODE: staleness check fails if line 4 has already been changed.
    VERIFIES WITH: grep DB_PATH SqliteDb.gd shows parachute_mutations.db.
    Source (Tier 1): Date (2003) "Introduction to Database Systems" Ch.1 p.7.
    Source (Tier 1): Stevens & Rago (2013) "APUE" Ch.3 p.55.
    Source (Tier 4): https://github.com/2shady4u/godot-sqlite/blob/master/src/gd_sqlite.cpp
    """
    if not SQLITEDB_GD.exists():
        print(f"  [ERROR] {SQLITEDB_GD} not found")
        return False

    content = SQLITEDB_GD.read_text(encoding="utf-8")
    old_line = 'const DB_PATH := "user://parachute.db"'
    new_line = f'const DB_PATH := "{DB_PATH}"'

    if new_line in content:
        print(f"  [SKIP] DB_PATH already points to parachute_mutations.db")
        return True

    if old_line not in content:
        # Check what's actually there
        for i, ln in enumerate(content.splitlines(), 1):
            if "DB_PATH" in ln:
                print(f"  [WARN] DB_PATH at line {i}: {ln!r}")
        print(f"  [ERROR] Expected line not found: {old_line!r}")
        return False

    new_content = content.replace(old_line, new_line, 1)

    ok, sha_before, sha_after = apply_with_structural_gate(
        conn, SESSION_ID, SQLITEDB_GD,
        lambda: SQLITEDB_GD.write_text(new_content, encoding="utf-8")
    )
    if not ok:
        return False

    # Verify
    written = SQLITEDB_GD.read_text(encoding="utf-8")
    if str(DB_PATH) not in written:
        print(f"  [ERROR] Read-back failed — DB_PATH not in written file")
        return False

    conn.execute("""
        INSERT OR IGNORE INTO gd_patch_log
            (session_id, file_path, patch_id_str, lines_changed,
             sha256_before, sha256_after, verified)
        VALUES (?,?,?,?,?,?,1)
    """, (SESSION_ID, str(SQLITEDB_GD), "F24-sqlitedb-absolute-path",
          1, sha_before, sha_after))
    conn.commit()
    print(f"  [OK] SqliteDb.gd DB_PATH -> {DB_PATH}")
    print(f"       sha_before={sha_before[:16]}  sha_after={sha_after[:16]}")
    log_orchestration(conn, "FIX_APPLIED", SQLITEDB_GD, "PASS")
    return True

def _read_proc_status(pid: int) -> dict:
    """
    WHAT: read /proc/PID/status for VmRSS and context switches.
    WHY:  resource_contention_log requires rss_kb, vol_ctxt_sw, nonvol_ctxt_sw.
    FAILURE MODE: process exited -> returns {}.
    Source (Tier 2): Linux procfs: https://www.kernel.org/doc/html/latest/filesystems/proc.html
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.5 p.208.
    """
    try:
        text = Path(f"/proc/{pid}/status").read_text()
        result = {}
        for ln in text.splitlines():
            if ln.startswith("VmRSS:"):
                result["vm_rss_kb"] = int(ln.split()[1])
            elif ln.startswith("voluntary_ctxt_switches:"):
                result["vol_ctxt_sw"] = int(ln.split()[1])
            elif ln.startswith("nonvoluntary_ctxt_switches:"):
                result["nonvol_ctxt_sw"] = int(ln.split()[1])
        return result
    except (FileNotFoundError, PermissionError):
        return {}


def _read_proc_stat() -> list:
    """
    WHAT: read /proc/stat cpu line for CPU/IO-wait calculation.
    WHY:  cpu_pct and io_wait_pct require two samples (delta).
    Source (Tier 2): https://www.kernel.org/doc/html/latest/filesystems/proc.html
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.6 p.258.
    """
    try:
        for ln in Path("/proc/stat").read_text().splitlines():
            if ln.startswith("cpu "):
                return [int(x) for x in ln.split()[1:]]
    except Exception:
        pass
    return [0] * 10


def compute_cpu_pct(prev: list, curr: list) -> tuple[float, float]:
    """
    WHAT: compute CPU% and IO-wait% from two /proc/stat samples.
    WHY:  USE method requires utilisation (CPU%) and saturation (IO wait%).
    Source (Tier 1): Gregg (2021) "Systems Performance" 2nd ed. Ch.2 p.58
      — USE method: Utilisation, Saturation, Errors.
    Source (Tier 2): https://www.kernel.org/doc/html/latest/filesystems/proc.html
    """
    if len(prev) < 8 or len(curr) < 8:
        return 0.0, 0.0
    # Fields: user nice system idle iowait irq softirq steal
    total_prev = sum(prev[:8])
    total_curr = sum(curr[:8])
    delta = total_curr - total_prev
    if delta == 0:
        return 0.0, 0.0
    idle_prev  = prev[3];  idle_curr  = curr[3]
    iowait_prev = prev[4]; iowait_curr = curr[4]
    cpu_pct    = 100.0 * (1.0 - (idle_curr - idle_prev) / delta)
    io_pct     = 100.0 * (iowait_curr - iowait_prev) / delta
    return round(cpu_pct, 1), round(io_pct, 1)


def run_ab_test_complete(conn: sqlite3.Connection,

        WHY:  the main thread is blocked on proc.stdout readline.

        nonlocal prev_stat
        # Own connection — WAL mode allows concurrent reads
        # Source (Tier 2): https://www.sqlite.org/wal.html

        while not stop_event.is_set():
            m = _read_proc_status(proc.pid)
            if not m:
                break
            curr = _read_proc_stat()
            cpu_pct, io_pct = compute_cpu_pct(prev_stat, curr)
            prev_stat = curr
            ts = datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ")
            try:
                mconn.execute("""
                    INSERT INTO resource_contention_log
                        (session_id, godot_pid, ts, vm_rss_kb,

                      m.get("nonvol_ctxt_sw"), cpu_pct, io_pct))
                mconn.commit()
            except Exception:
                pass
            if m.get("vm_rss_kb"):
