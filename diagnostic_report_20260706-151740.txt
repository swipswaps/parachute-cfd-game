    684	func _load_faa_obstacles() -> void:
    685		print("[VERBATIM] ENTER _load_faa_obstacles gate=none")
    686		var json_path: String = "res://data/faa_obstacles_kded_world.json"
    687		var f: FileAccess = FileAccess.open(json_path, FileAccess.READ)
    688		if not f:
    689			print("[VERBATIM] EXIT _load_faa_obstacles early=file_not_found path=", json_path)
    690			return
    691		var raw_text: String = f.get_as_text()
    692		f.close()
    693		print("[VERBATIM] _load_faa_obstacles read_bytes=", raw_text.length())
    694		var parsed = JSON.parse_string(raw_text)
    695		if parsed == null:
    696			print("[VERBATIM] EXIT _load_faa_obstacles early=json_parse_failed")
    697			return
    698		var obstacles: Array = parsed.get("obstacles", [])
    699		print("[VERBATIM] _load_faa_obstacles obstacle_count=", obstacles.size())
    700		var placed: int = 0
    701		for obs in obstacles:
    702			var wx: float = float(obs.get("world_x", 0.0))
    703			var wy: float = float(obs.get("ground_y", 0.0))
    704			var wz: float = float(obs.get("world_z", 0.0))
    705			var agl: float = float(obs.get("height_m", 5.0))
    706			var otype: String = str(obs.get("type", "UNKNOWN"))
    707			# WHAT: place a vertical cylinder at the obstacle position
    708			# WHY: cylinders are visible from altitude and represent towers/poles
    709			#   without requiring external assets.
    710			# SOURCE (Tier 2): Godot 4 CylinderMesh
    711			#   URL: https://docs.godotengine.org/en/stable/classes/class_cylindermesh.html
    712			#   VERBATIM: "height — Full height of the cylinder. Default value: 2.0"
    713			var mesh_inst: MeshInstance3D = MeshInstance3D.new()
    714			var cyl: CylinderMesh = CylinderMesh.new()
    715			cyl.height = agl
    716			if otype == "TOWER":
    717				cyl.top_radius = 0.5
    718				cyl.bottom_radius = 1.0
    719				if otype.begins_with("UTILITY") or otype == "POLE":
    720					cyl.top_radius = 0.15
    721					cyl.bottom_radius = 0.2
    722			else:
    723				cyl.top_radius = 0.3
    724				cyl.bottom_radius = 0.5
    725			cyl.radial_segments = 8
    726			mesh_inst.mesh = cyl
    727			# Position: centre of cylinder is at wy + agl/2 so base sits on ground
    728			mesh_inst.position = Vector3(wx, wy + agl * 0.5, wz)
    729			var mat: StandardMaterial3D = StandardMaterial3D.new()
    730			if otype == "TOWER":
    731				mat.albedo_color = Color(0.8, 0.8, 0.8)
    732			else:
    733				mat.albedo_color = Color(0.6, 0.5, 0.3)
    734			mat.shading_mode = BaseMaterial3D.SHADING_MODE_UNSHADED
    735			mesh_inst.material_override = mat
    736			add_child(mesh_inst)
    737			placed += 1
    738		print("[VERBATIM] FAA obstacles loaded: ", placed)
    739		print("[VERBATIM] EXIT _load_faa_obstacles ok=true placed=", placed)

    742	# Helper: find first MeshInstance3D child recursively

    745	func _find_first_mesh(node: Node) -> MeshInstance3D:

   1030	func _calculate_score():
   1031		print("[VERBATIM] ENTER _calculate_score gate=none")
   1032		var distance = _character.global_position.length()
   1033		var landing_speed = abs(_forward_speed) + abs(_descent_rate)
   1034		var distance_penalty = int(distance * 10)
   1035		var speed_penalty = int(landing_speed * 5) if landing_speed > 3.0 else 0
   1036		_score = 1000 - distance_penalty - speed_penalty
   1037		_score = max(0, _score)
   1038		print(
   1039			"[VERBATIM] Final score: ",
   1040			_score,
   1041			" (distance: ",
   1042			distance,
   1043			"m, landing speed: ",
   1044			landing_speed,
   1045			" m/s)"
   1046		)
   1047		_show_notification("Score: " + str(_score))
   1048		_score_label.text = "SCORE: " + str(_score)

   1355	# CFD wind update (called every physics frame)
   1356	# Ref: Derived from logarithmic wind profile theory (https://en.wikipedia.org/wiki/Logarithmic_wind_profile)
   1357	# ------------------------------------------------------------------
   1358	func _update_cfd_wind(delta: float):
   1359		var altitude_m = _current_altitude * 0.3048
   1360		var wind_speed_at_alt = _wind_base_speed * (1.0 + 0.1 * log(1.0 + altitude_m / 10.0))
   1361		_wind_gust_time += delta
   1362		if _wind_gust_time > randf_range(5.0, 10.0):
   1363			_wind_gust_time = 0.0
   1364			_wind_current_gust = randf_range(-_wind_turbulence, _wind_turbulence)
   1365		var final_speed = max(0.0, wind_speed_at_alt + _wind_current_gust)
   1366		var wind_rad = deg_to_rad(_wind_base_direction)
   1367		var wind_vec = Vector3(sin(wind_rad), 0, cos(wind_rad)) * (final_speed * 0.514444)
   1368		if _game_state == GameState.DIAGNOSIS:
   1369			_velocity_vec += wind_vec * delta * 0.5
   1370		print(
   1371			"[VERBATIM] CFD wind: speed ",
   1372			final_speed,
   1373			" kts, dir ",
   1374			_wind_base_direction,
   1375			"°, gust ",
   1376			_wind_current_gust
   1377		)

   1642	func _physics_process(delta):
   1643		if _game_state == GameState.IN_PLANE:
   1644			_plane_angle += _PLANE_ORBIT_SPEED * delta
   1645			if _plane_node:
   1646				_plane_node.position = Vector3(
   1647					cos(_plane_angle) * _PLANE_ORBIT_RADIUS,
   1648					_PLANE_ALTITUDE,
   1649					sin(_plane_angle) * _PLANE_ORBIT_RADIUS
   1650				)
   1651				_plane_node.rotation.y = _plane_angle + PI / 2.0
   1652			if _hud_labels.size() > 0:
   1653				_hud_labels[0].text = "ALT: 6000 ft (IN PLANE)"
   1654			if _hud_labels.size() > 7:
   1655				_hud_labels[7].text = "EP: Press J or SPACE to exit aircraft"
   1656			return
   1657		_prev_descent_rate = _descent_rate
   1658		_apply_malfunction_effects(delta)
   1659		var descent = _get_current_descent_rate() * 60.0 * delta
   1660		_character.position.y -= descent
   1661		if _character.position.y < 25.0:
   1662			_character.position.y = 25.0
   1663			if not _safe_landing:
   1664				ScreenshotLibrary.save_flight_screenshot()
   1665				print("[VERBATIM] FAILURE SCREENSHOT: ground impact (physics_process)")
   1666				_game_state = GameState.GAME_OVER
   1667				print("[VERBATIM] Ground impact – fatal")
   1668		_current_altitude = _character.position.y - 25.0
   1669	
   1670		_vario_mps = _prev_descent_rate - _descent_rate
   1671	
   1672		_update_cfd_wind(delta)
   1673		_update_canopy_tilt()
   1674	
   1675		if _game_state == GameState.FREEFALL:
   1676			var target_dir = -_character.global_position.normalized()
   1677			_forward_speed = move_toward(_forward_speed, _max_speed, _accel * delta)
   1678			var turn_dir = Vector3.RIGHT * _turn_input * _turn_force
   1679			_velocity_vec += turn_dir * delta
   1680			var forward_vec = target_dir * _forward_speed
   1681			_velocity_vec = _velocity_vec.move_toward(forward_vec, _accel * delta)
   1682			_character.position += _velocity_vec * delta
   1683			if _velocity_vec.length() > 0.5:
   1684				var angle = atan2(_velocity_vec.x, _velocity_vec.z)
   1685				_character.rotation = Vector3(0, angle, 0)
   1686			var speed_kts = _forward_speed * 1.94384
   1687			_hud_labels[1].text = "SPD: %.0f kts | VARIO: %+.1f m/s" % [speed_kts, _vario_mps]
   1688			_hud_labels[4].text = "TURN: %d" % (_turn_input * 100)
   1689			_check_decision_altitude()
   1690			# Capture flight screenshot every 5 seconds (R085 ensures during flight)
   1691			if _screenshot_save_timer > 0:
   1692				_screenshot_save_timer -= delta
   1693			if _screenshot_save_timer <= 0.0:
   1694				ScreenshotLibrary.save_flight_screenshot()
   1695				_screenshot_save_timer = 5.0
   1696	
   1697		if _game_state == GameState.OPENING_ANIM:

   1954	func _on_wind_received(
   1955		result: int, response_code: int, headers: PackedStringArray, body: PackedByteArray
   1956	):  # gdlint:ignore=unused-argument # gdlint:ignore=unused-argument # gdlint:ignore=unused-argument # gdlint:ignore=unused-argument # gdlint:ignore=unused-argument
   1957		if response_code != 200:

   1959			return
   1960		var json = JSON.parse_string(body.get_string_from_utf8())

   1976			return
   1977		var data = JSON.parse_string(body.get_string_from_utf8())
