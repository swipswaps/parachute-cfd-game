#!/usr/bin/env python3
"""
tag_and_audit_live_vs_backup.py
WHAT: adds an is_live_path column to component_registry (ALTER TABLE
      ADD COLUMN, additive only -- no existing column touched), sets it
      False for any relative_path containing the same backup-history
      markers already used by diff_all_tscn_backups.py earlier this
      session (old_copies, broken_backup, external_backups, .OFF,
      .disabled, .old, .pre_final, .pre_reconcile), then re-runs the
      self-collision query scoped to is_live_path=1 only.
WHY: the last run's "swipswaps/parachute-cfd-game (local)" self-
      collisions are very likely the same old_copies/.broken_backup
      duplicate addon trees already identified this session, ingested
      because teach_db_component_graph.py's rglob had no exclusion
      filter. Per Definition of Done, "taught" isn't met while genuine
      duplicates and known backup noise are indistinguishable in the
      data.
ASSUMES: run from parachute-cfd-game/ or reachable from it; the
         component_registry table from the prior run already exists.
VERIFIES WITH: real before/after row counts for is_live_path=1 vs 0,
               and the self-collision query re-run scoped to live-only,
               so any REMAINING duplicate is a genuine finding, not noise.
"""
import sqlite3
import sys
from pathlib import Path

EXCLUDE_MARKERS = [
    "old_copies", "broken_backup", "external_backups", "/BACKUP/",
    ".OFF", ".disabled", ".old", ".pre_final", ".pre_reconcile",
]

db_candidates = [c for c in Path(".").rglob("parachute_mutations.db") if c.stat().st_size > 0]
if not db_candidates:
    print("PRECONDITION FAILED: no non-empty parachute_mutations.db found", file=sys.stderr)
    sys.exit(1)
db_candidates.sort(key=lambda p: p.stat().st_size, reverse=True)
db_path = db_candidates[0]
print(f"=== Using: {db_path} ===\n")

conn = sqlite3.connect(str(db_path))
cur = conn.cursor()

cur.execute("PRAGMA table_info(component_registry)")
cols = [r[1] for r in cur.fetchall()]
if "is_live_path" not in cols:
    print("Adding is_live_path column (additive, ALTER TABLE ADD COLUMN only)...")
    cur.execute("ALTER TABLE component_registry ADD COLUMN is_live_path INTEGER")
    conn.commit()
else:
    print("is_live_path column already present.")

cur.execute("SELECT id, relative_path FROM component_registry WHERE source_repo LIKE '%local%'")
rows = cur.fetchall()
print(f"\n=== Tagging {len(rows)} local-project row(s) ===")
live_count, backup_count = 0, 0
for row_id, rel_path in rows:
    is_live = 0 if any(m in rel_path for m in EXCLUDE_MARKERS) else 1
    cur.execute("UPDATE component_registry SET is_live_path = ? WHERE id = ?", (is_live, row_id))
    if is_live:
        live_count += 1
    else:
        backup_count += 1
conn.commit()
print(f"  live: {live_count}  backup-history noise: {backup_count}")

print("\n=== Self-collisions WITHIN swipswaps/parachute-cfd-game, live-only ===")
cur.execute("""
    SELECT class_name, COUNT(*), GROUP_CONCAT(relative_path, ' || ')
    FROM component_registry
    WHERE source_repo LIKE '%local%' AND is_live_path = 1 AND class_name IS NOT NULL
    GROUP BY class_name HAVING COUNT(*) > 1
""")
real_dupes = cur.fetchall()
if real_dupes:
    print(f"  {len(real_dupes)} GENUINE live-tree duplicate(s) -- worth investigating each:")
    for name, n, paths in real_dupes:
        print(f"  {name} x{n}: {paths}")
else:
    print("  None -- prior 'duplicates' were entirely backup-history noise, now excluded.")

conn.close()
EOF
