class SelfHealThrottle:

        self._init_throttle_table()
    
    def _init_throttle_table(self):
        """Create throttle tracking table if it doesn't exist"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_path)
            cursor = conn.cursor()
            cursor.execute("""
                CREATE TABLE IF NOT EXISTS throttle_log (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    operation TEXT NOT NULL,
                    attempt_count INTEGER DEFAULT 0,
                    last_attempt TEXT,
                    backoff_seconds INTEGER DEFAULT 60,
                    success_count INTEGER DEFAULT 0,
                    failure_count INTEGER DEFAULT 0,
                    locked INTEGER DEFAULT 0,
                    UNIQUE(operation)
                )
            """)
            conn.commit()
            conn.close()
        except Exception as e:
            print(f"[WARN] Could not init throttle table: {e}")
    
    def can_attempt(self, operation: str) -> bool:
        """Check if we can attempt this operation based on backoff"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_path)
            cursor = conn.cursor()
            cursor.execute(
                "SELECT attempt_count, backoff_seconds, last_attempt, locked FROM throttle_log WHERE operation = ?",
                (operation,)
            )
            row = cursor.fetchone()
            conn.close()
            
            if not row:
                return True
            
            attempt_count, backoff_seconds, last_attempt, locked = row
            
            # If locked, don't attempt
            if locked:
                return False
            
            # Check if backoff time has elapsed
            if last_attempt:
                last_time = datetime.fromisoformat(last_attempt)
                elapsed = (datetime.now() - last_time).total_seconds()
                if elapsed < backoff_seconds:
                    return False
            
            return True
        except Exception as e:
            print(f"[WARN] Throttle check failed: {e}")
            return True  # Allow on failure
    
    def record_attempt(self, operation: str, success: bool):
        """Record an attempt and update backoff"""
        import sqlite3
        try:
            conn = sqlite3.connect(self.db_path)
            cursor = conn.cursor()
            
            # Get current state
            cursor.execute(
                "SELECT attempt_count, backoff_seconds, failure_count FROM throttle_log WHERE operation = ?",
                (operation,)
            )
            row = cursor.fetchone()
            
            if row:
                attempt_count, backoff_seconds, failure_count = row
                attempt_count += 1
                if success:
                    success_count = row[2] if len(row) > 2 else 0
                    # Reduce backoff on success
                    backoff_seconds = max(30, backoff_seconds // 2)
                    failure_count = 0
                else:
                    failure_count += 1
                    # Exponential backoff on failure
                    if failure_count > 3:

                    # Lock if too many failures
                    locked = 1 if failure_count > 5 else 0

                backoff_seconds = 30
                failure_count = 0 if success else 1
                locked = 0

            """, (
                operation,
                attempt_count,
                datetime.now().isoformat(),
                backoff_seconds,
                1 if success else 0,
                failure_count,
                locked
            ))
            conn.commit()
            conn.close()
            
            if locked and not success:
                print(f"[WARN] Operation '{operation}' locked due to repeated failures")
        except Exception as e:
            print(f"[WARN] Could not record attempt: {e}")
    
    def unlock(self, operation: str):

            )

    op["last_attempt"] = now
    op["attempts"] += 1

func execute_write(sql: String, args: Array = []) -> bool:
    """Execute a write operation with automatic recovery"""
    var result = execute_query(sql, args)
    return result != null and result.size() > 0

    _check_db_health()

func _ensure_db_ready() -> bool:
    """Ensure database is ready, attempt recovery if needed"""
    if not _db:
        _db = get_node_or_null("/root/SqliteDb")
        if not _db:
            _last_error = "SqliteDb not available"
            return false
    
    # Check if DB is actually working
    _check_db_health()
    
    if not _is_healthy:
        _attempt_recovery()
        return _is_healthy
    
    return true

func _check_db_health():
    """Check if database is functional"""
    if not _db:
        _is_healthy = false
        return
    
    if not _db.has_method("_query"):
        _is_healthy = false
        _last_error = "SqliteDb._query method not found"
        return
    
    # Test query
    var result = _db._query("SELECT 1")
    if result != null and result.size() > 0:
        _is_healthy = true
        _last_error = ""
        if _circuit_state == "OPEN" or _circuit_state == "HALF_OPEN":
            _circuit_state = "CLOSED"
            emit_signal("db_recovered")
    else:
        _is_healthy = false
        _last_error = "Health check failed"

func _attempt_recovery():
    """Attempt to recover database connection"""
    if _recovery_attempts >= _max_recovery_attempts:
        print("[VERBATIM] DatabaseGuard: max recovery attempts reached")
        return
    
    _recovery_attempts += 1
    print("[VERBATIM] DatabaseGuard: recovery attempt %d/%d" % [_recovery_attempts, _max_recovery_attempts])
    
    # Reload the database node
    if _db:
        _db.queue_free()
    
    _db = null
    OS.delay_msec(500)
    _db = get_node_or_null("/root/SqliteDb")
    
    if _db:
        _check_db_health()
        if _is_healthy:
            emit_signal("db_recovered")
            print("[VERBATIM] DatabaseGuard: recovery SUCCESS")
            return
    
    # If still unhealthy, schedule retry
    _schedule_retry()

func _schedule_retry():
    """Schedule a retry with exponential backoff"""
    var delay = min(_retry_delay_ms * pow(2, _connection_retries), 30000)
    _connection_retries += 1
    
    var timer = Timer.new()
    timer.wait_time = delay / 1000.0
    timer.one_shot = true
    timer.timeout.connect(_on_retry_timer_timeout)
    add_child(timer)
    timer.start()
    print("[VERBATIM] DatabaseGuard: retry scheduled in %d ms" % delay)

func _on_retry_timer_timeout():
    """Handle retry timer timeout"""
    _check_db_health()
    if not _is_healthy:
        _attempt_recovery()
