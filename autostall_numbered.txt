    50	    try:

    62	# Display provisioning: ensure a valid X11 display
    63	# ------------------------------------------------------------------
    64	def ensure_display() -> Tuple[bool, Optional[subprocess.Popen], str]:
    65	    """
    66	    Checks if DISPLAY is set and points to a valid X server.
    67	    If not, attempts to start Xvfb on a free display.
    68	    Returns (success, Xvfb_process, display_string).
    69	    """
    70	    display = os.environ.get("DISPLAY")
    71	    if display:
    72	        # Quick check: try to open a connection
    73	        try:
    74	            subprocess.check_call(["xdpyinfo", "-display", display], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    75	            print(f"[DISPLAY] Using existing display: {display}")
    76	            return True, None, display
    77	        except (subprocess.CalledProcessError, FileNotFoundError):
    78	            print(f"[DISPLAY] DISPLAY={display} is set but not usable. Will try to start Xvfb.")
    79	    else:
    80	        print("[DISPLAY] DISPLAY not set. Will try to start Xvfb.")
    81	
    82	    if not tool_available("Xvfb"):
    83	        print("[DISPLAY] Xvfb not installed. Falling back to headless (dummy renderer).")
    84	        print("[DISPLAY] To avoid grey screen, install Xvfb (e.g., 'sudo dnf install xorg-x11-server-Xvfb').")
    85	        return False, None, ""
    86	
    87	    # Try to find a free display number (we'll use :99 as default, but may try others)
    88	    for disp in [":99", ":100", ":101"]:
    89	        # Check if the display is already in use
    90	        lock_file = f"/tmp/.X{disp[1:]}-lock"
    91	        if os.path.exists(lock_file):
    92	            print(f"[DISPLAY] Display {disp} appears to be in use. Trying next.")
    93	            continue
    94	        try:
    95	            # Start Xvfb
    96	            proc = subprocess.Popen(
    97	                ["Xvfb", disp, "-screen", "0", XVFB_SCREEN, "-ac", "-nolisten", "tcp"],
    98	                stdout=subprocess.DEVNULL,
    99	                stderr=subprocess.DEVNULL
   100	            )
   101	            # Wait a moment for it to start
   102	            time.sleep(1)
   103	            # Verify it's running
   104	            if proc.poll() is None:
   105	                print(f"[DISPLAY] Started Xvfb on {disp}")
   106	                return True, proc, disp
   107	            else:
   108	                print(f"[DISPLAY] Xvfb on {disp} exited unexpectedly.")
   109	        except Exception as e:
   110	            print(f"[DISPLAY] Failed to start Xvfb on {disp}: {e}")
   111	            continue
   112	
   113	    print("[DISPLAY] Could not start Xvfb. Falling back to headless.")
   114	    return False, None, ""

   132	def extract_last_gd_error(output_buffer: List[str]) -> Tuple[Optional[str], Optional[int]]:
   133	    for line in reversed(output_buffer):
   134	        if "at:" in line and ".gd" in line:
   135	            gd_pos = line.find(".gd")
   136	            if gd_pos == -1:
   137	                continue
   138	            start = line.rfind(" ", 0, gd_pos) + 1
   139	            if start == 0:
   140	                start = line.rfind("(", 0, gd_pos) + 1
   141	            if start == 0:
   142	                start = 0
   143	            end = line.find(")", gd_pos)
   144	            if end == -1:
   145	                end = len(line)
   146	            candidate = line[start:end].strip()
   147	            if ":" in candidate:
   148	                file_part, num_part = candidate.rsplit(":", 1)
   149	                if num_part.isdigit():
   150	                    base = Path(PROJECT_DIR) / "scripts"
   151	                    file_name = file_part.split("/")[-1]
   152	                    full_path = base / file_name
   153	                    if not full_path.exists():
   154	                        full_path = file_part
   155	                    return str(full_path), int(num_part)
   156	        elif ".gd" in line and ":" in line:
   157	            match = re.search(r'\(([^)]+\.gd):(\d+)\)', line)
   158	            if match:
   159	                file_part = match.group(1)
   160	                line_num = int(match.group(2))
   161	                base = Path(PROJECT_DIR) / "scripts"
   162	                file_name = file_part.split("/")[-1]
   163	                full_path = base / file_name
   164	                if not full_path.exists():
   165	                    full_path = file_part
   166	                return str(full_path), line_num
   167	            match = re.search(r'([^\s(]+\.gd):(\d+)', line)
   168	            if match:
   169	                file_part = match.group(1)
   170	                line_num = int(match.group(2))
   171	                base = Path(PROJECT_DIR) / "scripts"
   172	                file_name = file_part.split("/")[-1]
   173	                full_path = base / file_name
   174	                if not full_path.exists():
   175	                    full_path = file_part
   176	                return str(full_path), line_num
   177	    return None, None

   180	# Proactive patch: add headless auto-start to build_terrain.gd (idempotent)
   181	# ------------------------------------------------------------------
   182	def apply_auto_start_patch() -> Tuple[bool, str]:
   183	    file_path = os.path.join(PROJECT_DIR, "scripts", "build_terrain.gd")
   184	    p = Path(file_path)
   185	    if not p.exists():
   186	        return False, f"File not found: {file_path}"
   187	    with open(p, 'r') as f:
   188	        lines = f.readlines()
   189	
   190	    # Check if the patch is already applied (look for the exact comment line)
   191	    for line in lines:
   192	        if "# Headless auto‑start: simulate SPACE press" in line:
   193	            return True, "Auto‑start patch already applied."
   194	
   195	    # Find the line with "Game ready – press SPACE at ~4000 ft to deploy"
   196	    ready_line_idx = None
   197	    for i, line in enumerate(lines):
   198	        if "Game ready – press SPACE at ~4000 ft to deploy" in line:
   199	            ready_line_idx = i
   200	            break
   201	    if ready_line_idx is None:
   202	        return False, "Could not locate 'Game ready' message line."
   203	
   204	    # Determine indentation of that line
   205	    indent = re.match(r'^[ \t]*', lines[ready_line_idx]).group(0)
   206	
   207	    # Create backup
   208	    backup_path = make_timestamped_backup(p)
   209	
   210	    # Insert auto-start block right after the message
   211	    auto_start_block = [
   212	        f"{indent}# Headless auto‑start: simulate SPACE press\n",
   213	        f"{indent}if OS.get_environment(\"GODOT_HEADLESS\") == \"1\":\n",
   214	        f"{indent}\tInput.action_press(\"ui_accept\")\n",
   215	        f"{indent}\tInput.action_release(\"ui_accept\")\n",
   216	        f"{indent}\tprint(\"[VERBATIM] Headless auto‑start triggered.\")\n",
   217	    ]
   218	    # Insert after the message line
   219	    lines[ready_line_idx+1:ready_line_idx+1] = auto_start_block
   220	
   221	    with open(p, 'w') as f:
   222	        f.writelines(lines)
   223	
   224	    return True, f"Auto‑start patch applied. Backup at {backup_path}"

   226	def make_timestamped_backup(p) -> object:
   227	    """Rule #21: creates a NEW backup file every call — never
   228	    overwrites a prior one. Microsecond precision so multiple
   229	    auto-fixes on the same file within the same run still each
   230	    get their own distinct backup."""
   231	    ts = datetime.now().strftime("%Y%m%d%H%M%S%f")
   232	    backup_path = p.with_suffix(p.suffix + f".bak.{ts}")
   233	    shutil.copy2(p, backup_path)
   234	    return backup_path

   236	# ------------------------------------------------------------------
   237	# Auto‑fix: indentation error
   238	# ------------------------------------------------------------------
   239	def auto_fix_indent_error(file_path: str, line_num: int) -> Tuple[bool, str]:
   240	    """
   241	    Fix an indentation error by checking the prevailing style in the
   242	    script's directory (learned from the codebase). Replaces the entire
   243	    line's leading whitespace with the correct number of tabs or spaces.
   244	    """
   245	    try:
   246	        p = Path(file_path)
   247	        if not p.exists():
   248	            return False, f"File not found: {file_path}"
   249	        with open(p, 'r') as f:
   250	            lines = f.readlines()
   251	        if line_num < 1 or line_num > len(lines):
   252	            return False, "Line number out of range"
   253	        offending = lines[line_num - 1]
   254	        stripped = offending.lstrip()
   255	        if not stripped:
   256	            return False, "Empty line - cannot fix"
   257	
   258	        # Determine the relative directory of this script
   259	        rel_dir = os.path.relpath(os.path.dirname(file_path), os.path.join(PROJECT_DIR, "scripts"))
   260	        if rel_dir == ".":
   261	            rel_dir = ""
   262	
   263	        style = get_indentation_style(rel_dir)
   264	        # Count current leading whitespace characters
   265	        leading = len(offending) - len(stripped)
   266	        if style == "tab":
   267	            # Assume 4 spaces = 1 tab for conversion
   268	            new_leading = "\t" * (leading // 4)
   269	        else:
   270	            new_leading = "    " * (leading // 4)   # space style
   271	        new_line = new_leading + stripped
   272	        # Ensure trailing newline
   273	        if not new_line.endswith("\n"):
   274	            new_line += "\n"
   275	
   276	        if new_line == offending:
   277	            return False, "No change needed (indentation already matches dominant style)"
   278	
   279	        backup_path = make_timestamped_backup(p)
   280	        lines[line_num - 1] = new_line
   281	        with open(p, 'w') as f:
   282	            f.writelines(lines)
   283	        return True, f"Indentation corrected to {style} style. Backup at {backup_path}"
   284	    except Exception as e:
   285	        return False, f"Error: {e}"

   287	def auto_fix_null_texture(file_path: str, line_num: int) -> Tuple[bool, str]:
   288	    try:
   289	        p = Path(file_path)
   290	        if not p.exists():
   291	            return False, "File not found"
   292	        with open(p, 'r') as f:
   293	            lines = f.readlines()
   294	
   295	        start_idx = None
   296	        for i in range(line_num - 1, -1, -1):
   297	            if "var tex" in lines[i] and "get_viewport" in lines[i]:
   298	                start_idx = i
   299	                break
   300	
   301	        if start_idx is None:
   302	            for i in range(line_num - 1, -1, -1):
   303	                if "get_viewport" in lines[i] and "get_texture" in lines[i]:
   304	                    start_idx = i
   305	                    break
   306	        if start_idx is None:
   307	            return False, "Could not locate the screenshot block (no var tex line)"
   308	
   309	        indent = re.match(r'^[ \t]*', lines[start_idx]).group(0)
   310	        end_idx = None
   311	        for i in range(start_idx + 1, len(lines)):
   312	            if lines[i].startswith(indent) and not lines[i].strip().startswith("elif") and not lines[i].strip().startswith("else"):
   313	                end_idx = i - 1
   314	                break
   315	        if end_idx is None:
   316	            end_idx = len(lines) - 1
   317	
   318	        backup_path = make_timestamped_backup(p)
   319	
   320	        # Replace with a comment and a print (no screenshot capture)
   321	        new_block = [
   322	            f"{indent}# Screenshot capture removed – rendering is working; no need to capture.\n",
   323	            f"{indent}print(\"[VERBATIM] Screenshot function removed (headless mode).\")\n"
   324	        ]
   325	        lines[start_idx:end_idx+1] = new_block
   326	
   327	        with open(p, 'w') as f:
   328	            f.writelines(lines)
   329	        return True, f"Screenshot block removed (no capture). Backup at {backup_path}"
   330	    except Exception as e:
   331	        return False, f"Error: {e}"

   337	def auto_fix_trailing_dot(file_path: str, line_num: int) -> Tuple[bool, str]:
   338	    """
   339	    If line `line_num` of `file_path` is a short line ending with a dot (like "dir."),
   340	    remove it completely. Returns (success, message).
   341	    """
   342	    try:
   343	        p = Path(file_path)
   344	        if not p.exists():
   345	            return False, f"File not found: {file_path}"
   346	        with open(p, 'r') as f:
   347	            original_lines = f.readlines()
   348	        if line_num < 1 or line_num > len(original_lines):
   349	            return False, "Line number out of range"
   350	        target = original_lines[line_num - 1].strip()
   351	        if target.endswith('.') and len(target) < 10:
   352	            # Create backup
   353	            backup_path = make_timestamped_backup(p)
   354	            # Remove the line
   355	            new_lines = original_lines[:line_num - 1] + original_lines[line_num:]
   356	            with open(p, 'w') as f:
   357	                f.writelines(new_lines)
   358	            return True, f"Removed stray dot line {line_num}: {target}"
   359	        else:
   360	            return False, "Line does not match stray‑dot pattern"
   361	    except Exception as e:
   362	        return False, f"Error: {e}"

   365	def rollback_file(file_path: str) -> bool:
   366	    """Restores from the MOST RECENT timestamped backup (Rule #21).
   367	    Filename format sorts lexicographically == chronologically, so
   368	    max() over the glob finds the latest one without parsing it."""
   369	    p = Path(file_path)
   370	    candidates = sorted(p.parent.glob(f"{p.name}.bak.*"))
   371	    if not candidates:
   372	        return False
   373	    latest_backup = candidates[-1]
   374	    shutil.copy2(latest_backup, p)
   375	    return True

   380	def print_source_block(filepath: str, lnum: int, context: int = CONTEXT_LINES):
   381	    p = Path(filepath)
   382	    if not p.exists():
   383	        print(f"      [WARNING] File not found: {filepath}")
   384	        return
   385	    lines = p.read_text().splitlines()
   386	    total = len(lines)
   387	    if lnum < 1 or lnum > total:
   388	        print(f"      [WARNING] line {lnum} out of range (file has {total} lines)")
   389	        return
   390	    start = max(0, lnum - context - 1)
   391	    end = min(total, lnum + context)
   392	    print(f"      [STALL SOURCE] {p} lines {start+1}-{end} (offending line {lnum}):")
   393	    for i in range(start, end):
   394	        prefix = ">> " if i == lnum - 1 else "   "
   395	        print(f"      {prefix}{i+1:4d}: {lines[i]}")

   397	# ------------------------------------------------------------------
   398	# Store error in database (auto‑schema detection)
   399	# ------------------------------------------------------------------
   400	def store_error(file_path: str, line_num: int, message: str, status: str = "new"):
   401	    try:
   402	        conn = sqlite3.connect(DB_PATH)
   403	        cursor = conn.cursor()
   404	        cursor.execute("SELECT name FROM sqlite_master WHERE type='table' AND name='files_to_fix'")
   405	        if cursor.fetchone():
   406	            sql = """
   407	                INSERT OR REPLACE INTO files_to_fix (file_path, line_number, error_text, status)
   408	                VALUES (?, ?, ?, ?)
   409	            """
   410	            cursor.execute(sql, (file_path, line_num, message, status))
   411	            conn.commit()
   412	            print(f"[DB] Stored/replaced error in files_to_fix with status {status}")
   413	        else:
   414	            cursor.execute('''
   415	                CREATE TABLE IF NOT EXISTS error_blocks (
   416	                    id INTEGER PRIMARY KEY AUTOINCREMENT,
   417	                    file_path TEXT NOT NULL,
   418	                    line_number INTEGER NOT NULL,
   419	                    error_text TEXT,
   420	                    context TEXT,
   421	                    status TEXT DEFAULT 'new',
   422	                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
   423	                    UNIQUE(file_path, line_number)
   424	                )
   425	            ''')
   426	            p = Path(file_path)
   427	            lines = p.read_text().splitlines() if p.exists() else []
   428	            start = max(0, line_num - CONTEXT_LINES - 1)
   429	            end = min(len(lines), line_num + CONTEXT_LINES)
   430	            context_lines = [f"{i+1:4d}: {lines[i]}" for i in range(start, end)] if lines else []
   431	            context_str = "\n".join(context_lines)
   432	            sql = """
   433	                INSERT OR REPLACE INTO error_blocks (file_path, line_number, error_text, context, status)
   434	                VALUES (?, ?, ?, ?, ?)
   435	            """
   436	            cursor.execute(sql, (file_path, line_num, lines[line_num-1].strip() if lines else message, context_str, status))
   437	            conn.commit()
   438	            print("[DB] Stored error in 'error_blocks'")
   439	        conn.close()
   440	    except Exception as e:
   441	        print(f"[DB] Error: {e}")

   443	# ------------------------------------------------------------------
   444	# Diagnostic tools (prioritized) – run only once
   445	# ------------------------------------------------------------------
   446	_diagnostics_run = False
   447	
   448	def run_diagnostics(pid: int) -> Dict[str, Any]:
   449	    global _diagnostics_run
   450	    if _diagnostics_run:
   451	        return {"skipped": "Diagnostics already run this session"}
   452	    _diagnostics_run = True
   453	
   454	    results = {}
   455	    proc_info = {}
   456	    try:
   457	        with open(f"/proc/{pid}/status", "r") as f:
   458	            proc_info["status"] = f.read()
   459	    except:
   460	        proc_info["status"] = "unavailable"
   461	    try:
   462	        with open(f"/proc/{pid}/wchan", "r") as f:
   463	            proc_info["wchan"] = f.read().strip()
   464	    except:
   465	        proc_info["wchan"] = "unavailable"
   466	    try:
   467	        with open(f"/proc/{pid}/cmdline", "rb") as f:
   468	            proc_info["cmdline"] = f.read().replace(b'\0', b' ').decode('utf-8')
   469	    except:
   470	        proc_info["cmdline"] = "unavailable"
   471	    results["proc_info"] = proc_info
   472	
   473	    if tool_available("gdb"):
   474	        results["gdb"] = run_cmd(["gdb", "-batch", "-p", str(pid), "-ex", "thread apply all bt"])
   475	    else:
   476	        results["gdb"] = "gdb not installed"
   477	
   478	    if tool_available("strace"):
   479	        results["strace"] = run_cmd(["timeout", str(TOOL_TIMEOUT), "strace", "-p", str(pid), "-s", "256"])
   480	    else:
   481	        results["strace"] = "strace not installed"
   482	
   483	    if tool_available("lsof"):
   484	        results["lsof"] = run_cmd(["lsof", "-p", str(pid)])
   485	    else:
   486	        results["lsof"] = "lsof not installed"
   487	
   488	    if tool_available("perf"):
   489	        record_cmd = ["perf", "record", "-g", "-p", str(pid), "-o", "/tmp/perf.data", "--", "sleep", str(TOOL_TIMEOUT)]
   490	        run_cmd(record_cmd, timeout=TOOL_TIMEOUT+1)
   491	        report_cmd = ["perf", "report", "-i", "/tmp/perf.data", "--stdio", "--max-stack", "10", "-n"]
   492	        results["perf"] = run_cmd(report_cmd, timeout=3)
   493	    else:
   494	        results["perf"] = "perf not installed"
   495	
   496	    if tool_available("ltrace"):
   497	        results["ltrace"] = run_cmd(["timeout", str(TOOL_TIMEOUT), "ltrace", "-p", str(pid), "-s", "256"])
   498	    else:
   499	        results["ltrace"] = "ltrace not installed"
   500	
   501	    if tool_available("iotop"):
   502	        results["iotop"] = run_cmd(["timeout", str(TOOL_TIMEOUT), "iotop", "-p", str(pid), "-b", "-n", "1"])
   503	    else:
   504	        results["iotop"] = "iotop not installed"
   505	
   506	    if tool_available("top"):
   507	        results["top"] = run_cmd(["top", "-b", "-n", "1", "-p", str(pid)])
   508	    else:
   509	        results["top"] = "top not installed"
   510	
   511	    if tool_available("sysdig"):
   512	        results["sysdig"] = run_cmd(["timeout", str(TOOL_TIMEOUT), "sysdig", "-p", "%proc.name %proc.pid %evt.type", "proc.pid=" + str(pid)])
   513	    else:
   514	        results["sysdig"] = "sysdig not installed"
   515	    if tool_available("vmstat"):
   516	        results["vmstat"] = run_cmd(["vmstat", "1", "2"])
   517	    else:
   518	        results["vmstat"] = "vmstat not installed"
   519	    if tool_available("iostat"):

   935	def _diag_if_error(line: str):
   936	    """Print diagnostic for any Godot error line that mentions a source file.
   937	       - .gd files: full source context (3‑5 lines).
   938	       - engine files: reads from local GODOT_SOURCE_DIR clone."""
   939	    import os
   940	
   941	    # 1. .gd files via robust extractor
   942	    gd_file, gd_line = extract_last_gd_error([line])
   943	    if gd_file and gd_line:
   944	        print(f"\n[DIAG] Error in {gd_file} at line {gd_line}:")
   945	        print_source_block(gd_file, gd_line)
   946	        return
   947	
   948	    # 2. Engine files referenced via "at:"
   949	    if "at:" in line:
   950	        start = line.find('(', line.find('at:') + 3)
   951	        if start != -1:
   952	            end = line.find(')', start)
   953	            if end != -1:
   954	                file_ref = line[start+1:end].strip()
   955	                if file_ref and ':' in file_ref:
   956	                    parts = file_ref.rsplit(':', 1)
   957	                    file_path = parts[0]
   958	                    line_num = int(parts[1]) if parts[1].isdigit() else None
   959	                    if line_num and not file_path.endswith('.gd'):
   960	                        local_path = os.path.join(GODOT_SOURCE_DIR, file_path)
   961	                        if os.path.exists(local_path):
   962	                            print(f"\n[DIAG] Engine error in {file_path} at line {line_num}:")
   963	                            # print context same way as print_source_block
   964	                            with open(local_path, "r", encoding="utf-8", errors="replace") as f:
   965	                                engine_lines = f.readlines()
   966	                            total = len(engine_lines)
   967	                            if 1 <= line_num <= total:
   968	                                start_idx = max(0, line_num - CONTEXT_LINES - 1)
   969	                                end_idx = min(total, line_num + CONTEXT_LINES)
   970	                                print(f"      [ENGINE SOURCE] {file_path} lines {start_idx+1}-{end_idx} (line {line_num}):")
   971	                                for i in range(start_idx, end_idx):
   972	                                    prefix = ">> " if i == line_num - 1 else "   "
   973	                                    print(f"      {prefix}{i+1:4d}: {engine_lines[i].rstrip()}")
   974	                            return
   975	                        else:
   976	                            print(f"\n[DIAG] Engine error in {file_path} at line {line_num} (source not found locally)")
   977	                            return
   978	                    elif line_num:
   979	                        # .gd file missed by extractor, try print_source_block anyway
   980	                        print(f"\n[DIAG] Error in {file_path} at line {line_num}:")
   981	                        print_source_block(file_path, line_num)
   982	                        return
   983	
   984	    # 3b. GDScript backtrace lines: "[N] func_name (res://path.gd:LINE)"
   985	    #     Always show source context — these carry exact file+line.
   986	    import re as _re
   987	    _bt = _re.match(r'\s*\[\d+\]\s+\S+\s+\((.+\.gd):(\d+)\)', line)
   988	    if _bt:
   989	        _f, _l = _bt.group(1), int(_bt.group(2))
   990	        _fp = _f.replace("res://", "godot_project/")
   991	        print(f"\n[DIAG] Backtrace: {_f} at line {_l}:")
   992	        print_source_block(_fp, _l)
   993	        return
   994	
   995	    # 3c. Engine-level "ERROR:" (HTTPRequest, physics, etc.)
   996	    if line.startswith("ERROR:") and ".gd" in line:
   997	        _parts = line.split(".gd")
   998	        if len(_parts) > 1:
   999	            _suffix = _parts[1]
  1000	            _colon = _suffix.find(":")
  1001	            if _colon != -1:
  1002	                try:
  1003	                    _ln2 = int(_suffix[_colon+1:].split()[0].rstrip(".,)"))
