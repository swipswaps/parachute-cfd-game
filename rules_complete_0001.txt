verify_insertion() {
    local decision="$1"
    local result=$(sqlite3 "$DB_PATH" "SELECT decision FROM file_audit WHERE file_path='$TARGET_FILE';")
    if [ "$result" = "$decision" ]; then
        log_result "verify_insertion" "true" "confirmed decision=$decision"
    else
        log_result "verify_insertion" "false" "mismatch: expected $decision, got $result"
    fi
}

show_context() {
    local line="$1"
    awk -v line="$line" \
        'NR>=line-3 && NR<=line+3 {
            prefix = (NR==line) ? ">>> " : "    "
            printf "%s%4d: %s\n", prefix, NR, $0
        }' "$TARGET_FILE"
}
