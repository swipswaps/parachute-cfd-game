CREATE TABLE pipelineruns (
            run_id TEXT PRIMARY KEY,
            started_at TEXT NOT NULL,
            ended_at TEXT,
            status TEXT NOT NULL,
            total_processed INTEGER DEFAULT 0,
            total_fixes INTEGER DEFAULT 0
        );
CREATE TABLE autohealactions (
            id TEXT PRIMARY KEY,
            run_id TEXT NOT NULL,
            iteration INTEGER NOT NULL,
            file_path TEXT NOT NULL,
            tool_used TEXT NOT NULL,
            success INTEGER NOT NULL,
            raw_output TEXT,
            timestamp TEXT NOT NULL, line_num INTEGER,
            FOREIGN KEY (run_id) REFERENCES pipelineruns(run_id) ON DELETE CASCADE
        );
CREATE TABLE verificationresults (
            id TEXT PRIMARY KEY,
            run_id TEXT NOT NULL,
            component TEXT NOT NULL,
            passed INTEGER NOT NULL,
            verified_at TEXT NOT NULL,
            FOREIGN KEY (run_id) REFERENCES pipelineruns(run_id) ON DELETE CASCADE
        );
CREATE TABLE sqlite_sequence(name,seq);
CREATE TABLE gd_patch_registry (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            file_path TEXT,
            line_num INTEGER,
            patch_type TEXT,
            description TEXT,
            applied_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        );
CREATE TABLE error_log (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            file_path TEXT,
            line_num INTEGER,
            error_type TEXT,
            message TEXT,
            captured_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            fixed INTEGER DEFAULT 0
        );
CREATE TABLE fix_strategies (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            error_type TEXT NOT NULL,
            description TEXT,
            pattern TEXT NOT NULL,
            replacement TEXT NOT NULL,
            validation_tool TEXT DEFAULT 'gdparse',
            priority INTEGER DEFAULT 0,
            enabled INTEGER DEFAULT 1,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        );
