import sys, datetime

def log_result(operation: str, success: bool, detail: str) -> None:
    ts = datetime.datetime.utcnow().isoformat()
    status = "SUCCESS" if success else "FAILURE"
    print(f"[{ts}] [{status}] {operation}: {detail}", file=sys.stderr)

import sqlite3

def confirm_tables(db_path: str) -> list:
    conn = sqlite3.connect(db_path)
    cur = conn.cursor()
    cur.execute("SELECT name FROM sqlite_master WHERE type='table'")
    tables = cur.fetchall()
    conn.close()
    if tables:
        log_result("confirm_tables", True, f"{len(tables)} table(s) found: {tables}")
    else:
        log_result("confirm_tables", False, "no tables found — do not assume schema")
    return tables

def diagnose(error_output: str) -> str:
    if "No such file or directory" in error_output:
        cause = "upstream step never created the expected file"
    elif "Permission denied" in error_output:
        cause = "execute bit or ownership mismatch, not a logic bug"
    else:
        cause = "unconfirmed — do not fix yet, gather more evidence"
    log_result("diagnose", cause != "unconfirmed — do not fix yet, gather more evidence", cause)
    return cause

def guarded_replace(text: str, old: str, new: str) -> str:
    count = text.count(old)
    if count != 1:
        log_result("guarded_replace", False, f"expected 1 match, found {count} — refusing")
        raise AssertionError(f"PRECONDITION VIOLATED: expected 1 match, found {count}")
    result = text.replace(old, new)
    log_result("guarded_replace", True, "exactly 1 match replaced")
    return result

import subprocess

def guarded_edit_and_verify(path: str, old: str, new: str, validate_cmd: list) -> dict:
    with open(path, "r") as f:
        text = f.read()

    count = text.count(old)
    if count != 1:
        log_result("guarded_edit_and_verify", False, f"expected 1 match, found {count}")
        return {"success": False, "reason": f"expected 1 match, found {count} — refusing to guess"}

    text = text.replace(old, new, 1)
    with open(path, "w") as f:
        f.write(text)

    with open(path, "r") as f:
        written = f.read()
    if written != text:
        log_result("guarded_edit_and_verify", False, "read-back mismatch after write")
        return {"success": False, "reason": "read-back does not match what was written"}

    result = subprocess.run(validate_cmd, capture_output=True, text=True)
    success = result.returncode == 0
    log_result("guarded_edit_and_verify", success,
               f"validator exit={result.returncode} stderr={result.stderr.strip() or '(empty)'}")
    return {
        "success": success,
        "exit_code": result.returncode,
        "stdout": result.stdout,
        "stderr": result.stderr,
    }

def write_and_confirm(path: str, new_content: str) -> bool:
    with open(path, "w") as f:
        f.write(new_content)
    with open(path, "r") as f:
        written = f.read()
    ok = written == new_content
    log_result("write_and_confirm", ok, "read-back matches" if ok else "read-back MISMATCH")
    if not ok:
        raise AssertionError("READ-AFTER-WRITE FAILED: written content does not match")
    return ok

def show_error_context(lines: list, error_line: int, context: int = 3) -> None:
    start, end = max(0, error_line - context), min(len(lines), error_line + context + 1)
    for i in range(start, end):
        prefix = ">>> " if i == error_line else "    "
        print(f"{prefix}{i+1:4d}: {lines[i].rstrip()}")
    log_result("show_error_context", True, f"showed lines {start+1}-{end}")

# Gating for complete fixes (generic template):
import shutil, subprocess

def run_gated_fix(target_path: str, find_issues, apply_fix, validate_cmd: list) -> int:
    backup_path = target_path + ".backup"
    issues = find_issues(target_path)
    if not issues:
        log_result("run_gated_fix", True, "no issues found")
        return 0

    shutil.copy2(target_path, backup_path)
    apply_fix(target_path, issues)

    result = subprocess.run(validate_cmd, capture_output=True, text=True)
    if result.returncode == 0:
        log_result("run_gated_fix", True, f"fixed {len(issues)} issue(s), verification passed")
        return 0

    shutil.copy2(backup_path, target_path)
    log_result("run_gated_fix", False,
               f"verification failed (exit {result.returncode}), restored from backup: {result.stderr.strip()}")
    return 1

# CODE:
def validate_rules_file(content: str, original_content: str, allowed_new_sections: list) -> dict:
    lines = content.splitlines()
    sections = [l.strip() for l in lines if l.strip().startswith("## ")]

    in_section, orphaned = False, []
    for i, line in enumerate(lines):
        if line.strip().startswith("## "):
            in_section = True
            continue
        if in_section and not line.strip():
            continue
        if not in_section and line.strip():
            orphaned.append((i + 1, line.strip()))

    if orphaned:
        log_result("validate_rules_file", False, f"orphaned text at lines: {[n for n, _ in orphaned]}")
        return {"valid": False, "reason": f"orphaned text outside any section header: {orphaned}"}

    original_sections = [l.strip() for l in original_content.splitlines() if l.strip().startswith("## ")]
    unexpected = [s for s in sections if s not in original_sections and s not in allowed_new_sections]
    if unexpected:
        log_result("validate_rules_file", False, f"unexpected sections: {unexpected}")
        return {"valid": False, "reason": f"unexpected new sections: {unexpected}"}

    missing = [s for s in original_sections if s not in sections]
    if missing:
        log_result("validate_rules_file", False, f"sections removed: {missing}")
        return {"valid": False, "reason": f"original sections removed: {missing}"}

    log_result("validate_rules_file", True, "structure clean")
    return {"valid": True, "reason": "only intended additions present"}

import subprocess, hashlib, shutil

def verified_restore(backup_path: str, target_path: str, expected_sha: str, validate_cmd: list) -> dict:
    actual_sha = hashlib.sha256(open(backup_path, "rb").read()).hexdigest()
    if actual_sha != expected_sha:
        log_result("verified_restore", False, f"checksum mismatch: expected {expected_sha}, got {actual_sha}")
        return {"success": False, "reason": "backup checksum mismatch — refusing to restore"}

    shutil.copy2(backup_path, target_path)
    result = subprocess.run(validate_cmd, capture_output=True, text=True)
    success = result.returncode == 0
    log_result("verified_restore", success,
               f"restored from {backup_path}, validator exit={result.returncode}")
    return {"success": success, "exit_code": result.returncode, "stdout": result.stdout, "stderr": result.stderr}

#!/usr/bin/env python3
"""
PRE-COMMIT HOOK: Validates every staged .sh and .py script against the rules.
Blocks the commit if any required function, header, or pattern is missing.
"""
import os, sys, subprocess, re

REQUIRED_BASH_FUNCTIONS = [
    r"init_db\s*\(\)",
    r"record_fix\s*\(\)",
    r"verify_insertion\s*\(\)",
]
REQUIRED_HEADER_PATTERN = r"# Citations \(verifiable\):"
REQUIRED_DRY_RUN_PATTERN = r"if \$DRY_RUN; then"
REQUIRED_VERIFIED_RESTORE = r"verified_restore|sha256"

FORBIDDEN_INVOCATIONS = [
    r"sed -i",
    r"2>/dev/null",
    r"set -e",
    r"grep -q",
]

def validate_script(filepath):
    with open(filepath, 'r') as f:
        content = f.read()

    # 1. Check for required DB functions (if script touches sqlite3)
    if 'sqlite3' in content:
        for func in REQUIRED_BASH_FUNCTIONS:
            if not re.search(func, content):
                return False, f"Missing required function: {func}"

    # 2. Check for the citation header
    if not re.search(REQUIRED_HEADER_PATTERN, content):
        return False, f"Missing header: '{REQUIRED_HEADER_PATTERN}'"

    # 3. Check for dry-run diagnostic parity (if script has a validate_cmd)
    if 'validate_cmd' in content and not re.search(REQUIRED_DRY_RUN_PATTERN, content):
        return False, f"Missing dry-run block: '{REQUIRED_DRY_RUN_PATTERN}'"

    # 4. Check for verified restore pattern
    if 'backup' in content and not re.search(REQUIRED_VERIFIED_RESTORE, content):
        return False, f"Missing verified restore (sha256/verified_restore)"

    # 5. STRICT BAN: actual invocations in executable lines (not comments)
    for pattern in FORBIDDEN_INVOCATIONS:
        # Search only non-comment lines
        for line in content.splitlines():
            if line.strip().startswith('#'):
                continue
            if re.search(pattern, line):
                return False, f"Forbidden invocation found: '{pattern}' in line: {line.strip()}"

    return True, "All checks passed."

def main():
    files = sys.argv[1:] if len(sys.argv) > 1 else subprocess.check_output(
        ['git', 'diff', '--cached', '--name-only', '--diff-filter=ACM']
    ).decode().splitlines()

    for f in files:
        if f.endswith(('.sh', '.py')):
            ok, msg = validate_script(f)
            if not ok:
                print(f"❌ {f}: {msg}")
                sys.exit(1)
            else:
                print(f"✅ {f}: {msg}")
    sys.exit(0)

if __name__ == "__main__":
    main()
